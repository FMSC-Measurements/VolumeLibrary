      IF(DBHOB .LE. 1.0)THEN

        RETURN
      ENDIF
      IF(HT1PRD .LE. 0.0) THEN

        RETURN
      ENDIF

      INDEX = ANINT(DBHOB) - 9

      LHT1 = HT1PRD / 10.0

      IF (INDEX.LT.1.OR.INDEX.GT.31) THEN
        VOL(2) = 0.0
      ELSEIF (LHT1 .EQ. 1) THEN
         VOL(2) = DOYAL(INDEX,1)   
      ELSEIF (LHT1 .EQ. 1.5) THEN

      ELSEIF (LHT1 .EQ. 6.0) THEN

      ENDIF

      VOL(3) = VOL(2)

      RETURN
