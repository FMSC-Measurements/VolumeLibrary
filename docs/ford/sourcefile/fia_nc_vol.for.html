C  This Subroutine searches an array to find the maching element
      SUBROUTINE SEARCH_SP(M,ARY,SPC,DONE,ERRFLAG)
      INTEGER M,I
      INTEGER ARY(M)
      INTEGER SPC,LOC,FIRST,LAST,HALF,DONE,ERRFLAG
      FIRST=1
      LAST=M
      DONE=0
      DO 5 I = FIRST, LAST
        IF(ARY(I).EQ.SPC)THEN
          DONE = I
          EXIT
        ENDIF
  5   CONTINUE 
  
      RETURN
      END
